cmake_minimum_required(VERSION 3.20)
project(rustveil_llvm LANGUAGES C CXX)

find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 20)
  message(FATAL_ERROR "This experiment requires LLVM 20")
endif()

add_executable(rustveil-llvm main.cpp)
target_compile_features(rustveil-llvm PRIVATE cxx_std_17)
target_include_directories(rustveil-llvm SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(LLVM_DEFINITIONS_LIST NATIVE_COMMAND "${LLVM_DEFINITIONS}")
target_compile_options(rustveil-llvm PRIVATE ${LLVM_DEFINITIONS_LIST})
llvm_map_components_to_libnames(LLVM_LIBS Core IRReader Support)
target_link_libraries(rustveil-llvm PRIVATE ${LLVM_LIBS})
if(MSVC)
  target_compile_options(rustveil-llvm PRIVATE /W4 /EHsc)
else()
  target_compile_options(rustveil-llvm PRIVATE -Wall -Wextra -Wpedantic)
endif()
